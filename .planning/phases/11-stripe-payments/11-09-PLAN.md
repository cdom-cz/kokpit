- R11-09-1 (research Schema item 3, Pitfall 12): columns `id` uuid v7; `type varchar(16)` (`card_payments_type_check`: matched, duplicate, unmatched, refund); `reason varchar(16)` (`card_payments_reason_check`: null for matched and refund, `already_paid` for duplicate, one of `amount`, `currency`, `cancelled`, `undetermined` for unmatched); `resolution varchar(16) DEFAULT 'none'` (`card_payments_resolution_check`: none, assigned, noted, returned; `card_payments_resolution_type_check`: matched only none, refund only none or noted, assigned and returned only on duplicate or unmatched); `resolved_at`, `resolved_by` (users), `resolution_note varchar(500)` (`card_payments_resolved_check`: resolved_at and resolved_by present exactly when the resolution is not none; `card_payments_note_check`: noted needs a non-blank note); `assigned_invoice_id` (invoices; present exactly when assigned); `amount_minor bigint > 0`, `currency char(3)` regex; `offered_amount_minor bigint` and `offered_currency char(3)`, both or neither (the amount the link offered, for the "Nabízeno" line); `stripe_event_id varchar(255) NOT NULL` unique; `stripe_session_id` partial unique where not null; `stripe_payment_intent_id` partial unique `WHERE type <> 'refund' AND stripe_payment_intent_id IS NOT NULL`; `stripe_refund_id` unique, present exactly for refunds (`card_payments_refund_id_check`); `stripe_link_id varchar(255)`; `invoice_id` (invoices, NO ACTION; required for matched, `card_payments_matched_invoice_check`); `payment_link_id` (payment_links); `invoice_payment_id` unique with `card_payments_invoice_payment_fk ...
